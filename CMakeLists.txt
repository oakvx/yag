cmake_minimum_required(VERSION 3.16)
project(yag VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_executable(yag host/src/main.c host/src/http/server.c host/src/inference/openai.c host/src/core/config.c host/src/protocols/openai_responses/request.c host/src/protocols/openai_responses/response.c host/src/protocols/anthropic/messages.c third_party/yyjson/yyjson.c)
target_include_directories(yag PRIVATE host/include third_party/yyjson)
target_link_libraries(yag PRIVATE CURL::libcurl Threads::Threads)
target_compile_definitions(yag PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(yag PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:yag>)
endif()
if(Python3_Interpreter_FOUND)
 add_test(NAME responses COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/responses.py $<TARGET_FILE:yag>)
endif()

if(Python3_Interpreter_FOUND)
 add_test(NAME rate_limit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/rate_limit.py $<TARGET_FILE:yag>)
endif()

if(Python3_Interpreter_FOUND)
 add_test(NAME anthropic COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/anthropic.py $<TARGET_FILE:yag>)
endif()

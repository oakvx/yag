cmake_minimum_required(VERSION 3.16)
project(yag VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
find_package(CURL REQUIRED)
find_package(Threads REQUIRED)
add_executable(yag host/src/main.c host/src/http/server.c host/src/inference/openai.c host/src/inference/models.c host/src/inference/health.c host/src/inference/provider_status.c host/src/inference/model_capabilities.c host/src/core/config.c host/src/core/scheduler.c host/src/core/client_registry.c host/src/core/admin_audit.c host/src/core/system_resources.c host/src/inference/ollama_admin.c host/src/protocols/openai_responses/request.c host/src/protocols/openai_responses/response.c host/src/protocols/anthropic/messages.c host/src/protocols/anthropic/stream.c third_party/yyjson/yyjson.c)
target_include_directories(yag PRIVATE host/include third_party/yyjson)
target_link_libraries(yag PRIVATE CURL::libcurl Threads::Threads)
target_compile_definitions(yag PRIVATE _POSIX_C_SOURCE=200809L)
target_compile_options(yag PRIVATE -Wall -Wextra -Wpedantic)
